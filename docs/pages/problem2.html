wide_daily_returns <- daily_data_rescaled_price %>%
  group_by(symbol) %>%
  arrange(date) %>%
  select(date, symbol, daily_percent_gain) %>%
  mutate(symbol = case_when(
    symbol == "^GSPC" ~ "SP500",
    symbol == "GOOG" ~ "GOOG",
    symbol == "NVDA" ~ "NVDA"
  )) %>%
  pivot_wider(names_from = symbol, values_from = daily_percent_gain) %>%
  filter(complete.cases(.))
wide_daily_returns %>%
  ggplot(aes(x=SP500, y=GOOG))+
  geom_point()+
  labs(title= "Daily Change in SP500 and Google Returns",
       x="SP500 Daily Change Percentage",
       y="Google Daily Change Percentage")+
  theme_tq()
cor_daily_matrix <- wide_daily_returns %>%
  select(SP500, GOOG, NVDA) %>%
  cor()
